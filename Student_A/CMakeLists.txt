cmake_minimum_required(VERSION 3.20)
project(GraphPaths LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(GraphPathsStatic STATIC src/GraphPaths.cpp)
target_compile_definitions(GraphPathsStatic
    PUBLIC GRAPHPATHS_STATIC
)
target_include_directories(GraphPathsStatic PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_library(GraphPaths SHARED src/GraphPaths.cpp)
target_compile_definitions(GraphPaths PRIVATE GRAPH_PATHS_EXPORTS)
target_include_directories(GraphPaths PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(StaticTest tests/main.cpp)
target_link_libraries(StaticTest PRIVATE GraphPathsStatic)
add_executable(ImplicitSelfTest tests/main.cpp)
target_link_libraries(ImplicitSelfTest PRIVATE GraphPaths)
if(WIN32)
add_custom_command(TARGET ImplicitSelfTest POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:GraphPaths> $<TARGET_FILE_DIR:ImplicitSelfTest>)
endif()
